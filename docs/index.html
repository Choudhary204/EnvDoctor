    🚀 Quick Start Guide
    
        
    
          Install EnvDoctor in your Node.js projects or run it instantly with npx.
        
    

        
    1. Run with npx (No install needed)
    
        
    # Health check your default .env
npx @choudhary204/env-doctor check

# Check with required variables & custom file
npx @choudhary204/env-doctor check --file .env.production --required PORT,DATABASE_URL,JWT_SECRET

# Compare development vs production
npx @choudhary204/env-doctor compare .env.development .env.production

# Generate a safe .env.example
npx @choudhary204/env-doctor generate-example --file .env --output .env.example
    

        
    2. Install as dependency
    
        
    # Install locally
npm install @choudhary204/env-doctor

# Or install globally for CLI command
npm install -g @choudhary204/env-doctor
    

        
    3. Use in your Node.js code
    
        
    import { diagnose } from '@choudhary204/env-doctor';

const report = await diagnose({
  filePath: '.env',
  required: ['PORT', 'DATABASE_URL', 'JWT_SECRET'],
  rules: {
    PORT: 'port',
    DATABASE_URL: 'url',
    JWT_SECRET: { type: 'string', minLength: 32 },
    NODE_ENV: { enum: ['development', 'production', 'test'] },
    DEBUG: 'boolean',
  },
  checkGit: true,
});

console.log('Health Score:', report.healthScore); // e.g. 95
console.log('Errors:', report.errors);
console.log('Fixes:', report.suggestedFixes);
